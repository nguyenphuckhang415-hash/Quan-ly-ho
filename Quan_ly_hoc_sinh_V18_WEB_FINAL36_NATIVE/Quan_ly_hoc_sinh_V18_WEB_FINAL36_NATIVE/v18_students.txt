    def delete_one_student(self, tree):
        sel = tree.selection()
        if not sel:
            messagebox.showwarning("Chọn", "Hãy chọn học sinh cần xóa.")
            return

        values = tree.item(sel[0])["values"]
        student_id = values[0]
        student_name = values[1]

        password = simpledialog.askstring(
            "🔐 Xác nhận xóa",
            "Nhập mật khẩu giáo viên:",
            show="*"
        )
        if password is None:
            return

        con = connect()
        row = con.execute(
            "SELECT password_hash FROM teacher WHERE id=1"
        ).fetchone()

        if not row or row["password_hash"] != hash_pw(password):
            con.close()
            messagebox.showerror("Từ chối", "Mật khẩu giáo viên không đúng.")
            return

        ok = messagebox.askyesno(
            "Xóa toàn bộ dữ liệu học sinh",
            f'Xóa "{student_name}" và toàn bộ dữ liệu phụ huynh, điểm, nhiệm vụ, mã xác nhận?'
        )
        if not ok:
            con.close()
            return

        con.execute("DELETE FROM students WHERE id=?", (student_id,))
        con.commit()
        con.close()
        messagebox.showinfo("Đã xóa", "Đã xóa hoàn toàn dữ liệu học sinh.")
        self.manage_students()

                    positions.append((cx-rx*math.cos(ang), cy-ry*math.sin(ang)))
            else:
                cols=max(3,min(6,(w-80)//170));
                for i in range(n):
                    row=i//cols; col=i%cols
                    positions.append((90+col*170,160+row*100))
            for row,(x,y) in zip(r,positions):
                fill="#fff7e6" if row["role"] else "#f7fbff"
                outline="#f39c12" if row["role"] else "#7aa2e8"
                item=canvas.create_rectangle(x,y,x+145,y+72,fill=fill,outline=outline,width=2)
                title=row["name"][:20]
                role_text=f"\n{row['role']}" if row["role"] else ""
                canvas.create_text(x+72,y+22,text=title,font=("Arial",10,"bold"),fill="#263238")
                canvas.create_text(x+72,y+48,text=f"{row['team']} • {row['role']}" if row['role'] else row['team'],font=("Arial",9),fill="#667085")
                node_map[item]=(row,x,y)
                canvas.tag_bind(item,"<Button-1>",lambda e,it=item: click_node(it))
                if teacher_edit:
                    canvas.tag_bind(item,"<B1-Motion>",lambda e,it=item: drag_node(e,it))
                    canvas.tag_bind(item,"<ButtonPress-1>",lambda e,it=item: start_drag(e,it))

        def click_node(item):
            row=node_map[item][0]
            info.set(f"{row['name']}  •  {row['class_name']}  •  {row['team']}  •  {row['role'] or 'Học sinh'}")

        def start_drag(event,item):
            drag["item"]=item; drag["offset"]=(event.x, event.y)
        def drag_node(event,item):

            row,x,y=node_map[item]; dx=event.x-drag["offset"][0]; dy=event.y-drag["offset"][1]

        layout_box.bind("<<ComboboxSelected>>",lambda e: draw())

    # -------------------- RANKING --------------------
    def ranking(self):
        self.clear()
        self.header("🏆 XẾP HẠNG HỌC SINH", "Tổng điểm cộng / trừ")

        con = connect()
        rows = con.execute("""
            SELECT s.name,s.class_name,s.team,
                   COALESCE(SUM(sc.points),0) AS total
            FROM students s
            LEFT JOIN scores sc ON s.id=sc.student_id
            WHERE s.status IN ('approved','verified')
            GROUP BY s.id
            ORDER BY total DESC, s.name ASC
        """).fetchall()
        con.close()

        tr = self.tree(
            ("rank","name","class","team","total"),
            ("Hạng","Họ tên học sinh","Lớp","Tổ","Tổng điểm"),
            [70,320,100,90,140]
        )
        for i, r in enumerate(rows, start=1):
            tr.insert("", "end", values=(
                i, r["name"], r["class_name"], r["team"], r["total"]
            ))

        self.button(
            self, "Quay lại", self.teacher_dashboard, 15, "#7f8c8d"
        ).pack(pady=8)

    # -------------------- ACCOUNT / SECURITY MANAGEMENT --------------------
    def manage_accounts(self):
        self.clear()
        self.header("🔐 TÀI KHOẢN & BẢO MẬT", "Giáo viên xem trạng thái tài khoản và quản lý quyền truy cập")

        con = connect()
        rows = con.execute("""
            SELECT p.student_id, p.email, p.verified, p.must_change,
                   s.name, s.class_name, s.team, s.parent_name, s.status
            FROM parents p JOIN students s ON s.id=p.student_id
            ORDER BY s.class_name, s.name
        """).fetchall()
        con.close()

        tr = self.tree(
            ("sid","student","class","team","parent","email","verified","must_change"),
            ("ID HS","Học sinh","Lớp","Tổ","Phụ huynh","Email tài khoản PH","Xác nhận","Đổi MK lần đầu"),
            [70,200,90,70,180,280,100,120]
        )
        for r in rows:
            tr.insert("", "end", values=(
                r["student_id"], r["name"], r["class_name"], r["team"],
                r["parent_name"], r["email"],
                "Đã xác nhận" if r["verified"] else "Chưa xác nhận",
                "Có" if r["must_change"] else "Không"
            ))

        panel = tk.Frame(self, bg=COLORS["bg"])
        panel.pack(pady=8)

        def reset_parent():
            sel = tr.selection()
            if not sel:
                messagebox.showwarning("Chọn", "Hãy chọn tài khoản phụ huynh.")
                return
            sid = tr.item(sel[0])["values"][0]
            teacher_pw = simpledialog.askstring("Xác nhận giáo viên", "Nhập mật khẩu giáo viên:", show="*")
            if teacher_pw is None:
                return
            con2 = connect()
            row = con2.execute("SELECT password_hash FROM teacher WHERE id=1").fetchone()
            if not row or row["password_hash"] != hash_pw(teacher_pw):
                con2.close()
                messagebox.showerror("Từ chối", "Mật khẩu giáo viên không đúng.")
                return
            temp = make_parent_temp()
            con2.execute("UPDATE parents SET password_hash=?, password_display=?, must_change=1 WHERE student_id=?", (hash_pw(temp), temp, sid))
            con2.commit(); con2.close()
            messagebox.showinfo("Đã cấp lại", f"Mật khẩu phụ huynh mới do giáo viên cấp:\n{temp}\n\nĐây không phải mật khẩu email.")
            self.manage_accounts()

        def delete_parent():
            sel = tr.selection()
            if not sel:
                messagebox.showwarning("Chọn", "Hãy chọn phụ huynh.")
                return
            sid = tr.item(sel[0])["values"][0]
            teacher_pw = simpledialog.askstring("Xóa phụ huynh", "Nhập mật khẩu giáo viên:", show="*")
            if teacher_pw is None:
                return
            con2 = connect()
            row = con2.execute("SELECT password_hash FROM teacher WHERE id=1").fetchone()
            if not row or row["password_hash"] != hash_pw(teacher_pw):
                con2.close(); messagebox.showerror("Từ chối", "Mật khẩu giáo viên không đúng."); return
            ok = messagebox.askyesno("Xóa tài khoản phụ huynh", "Xóa tài khoản phụ huynh khỏi hệ thống? Hồ sơ học sinh vẫn giữ nguyên.")
            if ok:
                con2.execute("DELETE FROM parents WHERE student_id=?", (sid,))
                con2.commit()
            con2.close()
            if ok: self.manage_accounts()

        tk.Label(panel, text="Lưu ý: vì lý do bảo mật, mật khẩu thật không hiển thị. Giáo viên có thể cấp lại mật khẩu.", bg=COLORS["bg"], fg=COLORS["gray"]).pack(side="left", padx=8)
        self.button(panel, "🔁 Cấp lại MK PH", reset_parent, 16, COLORS["purple"]).pack(side="left", padx=4)
        self.button(panel, "🗑 Xóa PH", delete_parent, 13, COLORS["red"]).pack(side="left", padx=4)
        self.button(self, "Quay lại", self.teacher_dashboard, 15, "#7f8c8d").pack(pady=8)

            con.close()
            messagebox.showerror("Từ chối", "Mật khẩu giáo viên không đúng.")
