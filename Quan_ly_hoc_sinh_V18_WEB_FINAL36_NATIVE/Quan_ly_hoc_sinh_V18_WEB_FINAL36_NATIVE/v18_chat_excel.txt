    def chat_reminder_dialog(self):
        win=tk.Toplevel(self); win.title("⏰ Nhắc hẹn - chỉ giáo viên"); win.geometry("520x300")
        fields=[]
        for i,(lab,default) in enumerate([("Tiêu đề","Nhắc kiểm tra nhiệm vụ"),("Thời gian",""),("Nội dung","")]):
            tk.Label(win,text=lab+":",font=("Arial",10,"bold")).grid(row=i,column=0,sticky="e",padx=10,pady=9)
            e=tk.Entry(win,width=45); e.insert(0,default); e.grid(row=i,column=1,padx=10,pady=9); fields.append(e)
        fields[1].insert(0,datetime.now().strftime("%Y-%m-%d %H:%M"))
        def save():
            title,when,note=[e.get().strip() for e in fields]
            try: datetime.strptime(when,"%Y-%m-%d %H:%M")
            except ValueError: messagebox.showerror("Thời gian","Dùng YYYY-MM-DD HH:MM"); return
            if not title:return
            con=connect(); con.execute("INSERT INTO chat_reminders(title,remind_at,note,created_at) VALUES(?,?,?,?)",(title,when,note,now())); con.commit(); con.close(); win.destroy(); self._schedule_reminder(title,when,note); messagebox.showinfo("Đã đặt nhắc hẹn",f"{title}\n{when}")
        self.button(win,"💾 Lưu nhắc hẹn",save,16,COLORS["pink"]).grid(row=3,column=0,columnspan=2,pady=12)
        win.grab_set()

    def _schedule_reminder(self,title,when,note=""):
        try:
            dt=datetime.strptime(when,"%Y-%m-%d %H:%M")
            delay=max(1000,int((dt-datetime.now()).total_seconds()*1000))
            self.after(delay,lambda:messagebox.showinfo("⏰ Nhắc hẹn giáo viên",title+(f"\n{note}" if note else "")))
        except Exception: pass

    def import_students_excel(self):
        if not OPENPYXL_OK:
            messagebox.showerror("Thiếu thư viện", "Chưa có openpyxl. Hãy cài bằng: pip install openpyxl")
            return
        f=filedialog.askopenfilename(title="Tải danh sách học sinh từ Excel",filetypes=[("Excel","*.xlsx *.xlsm"), ("Tất cả","*.*")])
        if not f: return
        try:
            wb=load_workbook(f,data_only=True); ws=wb.active
            rows=list(ws.iter_rows(values_only=True))
            if not rows: raise ValueError("File Excel không có dữ liệu.")

            # Hỗ trợ bảng có tiêu đề nhiều dòng như mẫu danh sách trường:
            # dòng 1-5 là tiêu đề hành chính, dòng 6 mới là hàng cột.
            def norm_header(v):
                if v is None: return ""
                return " ".join(str(v).replace("\n"," ").split()).strip().lower()

            aliases={
                "họ tên học sinh":"name", "họ tên":"name", "tên học sinh":"name", "name":"name",
                "lớp":"class_name", "lớp học":"class_name", "class":"class_name",
                "tổ":"team", "tổ/nhóm":"team", "nhóm":"team", "team":"team",
                "giáo viên chủ nhiệm":"homeroom_teacher", "gv chủ nhiệm":"homeroom_teacher", "gvcnh":"homeroom_teacher",
                "phụ huynh":"parent_name", "họ tên phụ huynh":"parent_name",
                "email phụ huynh":"parent_email", "email":"parent_email"
            }

            header_row=None; idx={}
            # Tìm hàng chứa cột bắt buộc trong tối đa 20 dòng đầu.
            for ri, rr in enumerate(rows[:20]):
                test={aliases[norm_header(x)]:i for i,x in enumerate(rr) if norm_header(x) in aliases}
                if "name" in test:
                    header_row=ri; idx=test; break
            if header_row is None or "name" not in idx:
                raise ValueError("Không tìm thấy cột 'Họ tên học sinh'. Hãy kiểm tra hàng tiêu đề trong file Excel.")

            # Nếu file không có cột Lớp, tự lấy tên lớp từ tiêu đề, ví dụ
            # 'DANH SÁCH HỌC SINH LỚP 8/4'.
            inferred_class=""
            import re as _re
            for rr in rows[:header_row]:
                joined=" ".join(norm_header(x) for x in rr if x is not None)
                m=_re.search(r"danh sách học sinh\s+lớp\s+(.+)$", joined, _re.I)
                if m:
                    inferred_class=m.group(1).strip().rstrip(".")
                    break
            if "class_name" not in idx and not inferred_class:
                raise ValueError("Cần cột 'Lớp' hoặc tiêu đề file phải ghi tên lớp, ví dụ 'DANH SÁCH HỌC SINH LỚP 8/4'.")

            added=0; skipped=0; con=connect()
            for rr in rows[header_row+1:]:
                def val(key, default=""):
                    i=idx.get(key); return str(rr[i]).strip() if i is not None and i < len(rr) and rr[i] is not None else default
                name=val("name"); cls=val("class_name", inferred_class); team=val("team",""); teacher=val("homeroom_teacher",TEACHER_NAME); parent=val("parent_name",""); email=val("parent_email","")
                if not name or not cls: continue
                if con.execute("SELECT 1 FROM students WHERE lower(name)=lower(?) AND class_name=?",(name,cls)).fetchone(): skipped+=1; continue
                next_id=con.execute("SELECT COALESCE(MAX(id),0)+1 AS n FROM students").fetchone()["n"]
                su=f"hs{int(next_id):04d}"; sp=secrets.token_hex(3); pu=f"ph{int(next_id):04d}"; pp=secrets.token_hex(3)
                con.execute("INSERT INTO students(name,class_name,team,homeroom_teacher,parent_name,parent_email,group_name,status,access_code,student_username,student_password_hash,student_password_display,created_at,approved_at,verified_at) VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)",(name,cls,team,teacher,parent,email,"","approved",make_code(),su,hash_pw(sp),sp,now(),now(),now()))
                sid=con.execute("SELECT last_insert_rowid() AS id").fetchone()["id"]
                con.execute("INSERT OR REPLACE INTO parents(student_id,email,username,password_hash,password_display,must_change,verified) VALUES(?,?,?,?,?,?,1)",(sid,email,pu,hash_pw(pp),pp,0))
                con.execute("INSERT OR IGNORE INTO classes(class_name,homeroom_teacher,group_name) VALUES(?,?,?)",(cls,teacher,f"Nhóm lớp {cls}"))
                added+=1
            con.commit(); con.close()
            messagebox.showinfo("Nhập Excel thành công",f"Đã thêm {added} học sinh. Bỏ qua {skipped} học sinh trùng.")
            self.manage_students()
        except Exception as e:
            try: con.close()
            except Exception: pass
            messagebox.showerror("Nhập Excel thất bại",str(e))
