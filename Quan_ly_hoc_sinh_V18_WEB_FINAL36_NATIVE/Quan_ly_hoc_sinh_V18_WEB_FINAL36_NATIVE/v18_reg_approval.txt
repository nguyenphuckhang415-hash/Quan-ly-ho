    def student_register(self):
        self.clear()
        self.header(
            "🎓 ĐĂNG KÝ HỌC SINH",
            "Học sinh tự tạo tài khoản + mật khẩu → gửi giáo viên chủ nhiệm → chờ phê duyệt"
        )

        card = tk.Frame(self, bg="white", bd=1, relief="solid")
        card.pack(pady=10, ipadx=35, ipady=20)

        labels = [
            "Họ tên học sinh",
            "Tên đăng nhập học sinh",
            "Mật khẩu học sinh",
            "Lớp",
            "Tổ",
            "Giáo viên chủ nhiệm",
            "Họ tên phụ huynh",
            "Email phụ huynh",
        ]
        entries = []

        for i, label in enumerate(labels):
            tk.Label(card,text=label+":",bg="white",font=("Arial",10,"bold")).grid(
                row=i,column=0,sticky="e",padx=10,pady=7
            )
            e=tk.Entry(card,width=42,show="*" if "Mật khẩu" in label else "")
            e.grid(row=i,column=1,padx=10,pady=7)
            entries.append(e)

        entries[5].insert(0, TEACHER_NAME)

        def submit():
            values=[e.get().strip() for e in entries]
            if not all(values):
                messagebox.showwarning(
                    "Thiếu thông tin",
                    "Hãy nhập đầy đủ họ tên, tên đăng nhập, mật khẩu, lớp, tổ, "
                    "giáo viên chủ nhiệm, họ tên phụ huynh và email."
                )
                return
            if len(values[2]) < 6:
                messagebox.showwarning("Mật khẩu","Mật khẩu học sinh phải có ít nhất 6 ký tự.")
                return
            if "@" not in values[7]:
                messagebox.showwarning("Email","Email phụ huynh chưa hợp lệ.")
                return

            con=connect()
            try:
                exists=con.execute(
                    "SELECT 1 FROM students WHERE lower(student_username)=lower(?) LIMIT 1",
                    (values[1],)
                ).fetchone()
                if exists:
                    messagebox.showwarning("Tên đăng nhập","Tên đăng nhập học sinh đã tồn tại.")
                    return

                con.execute("""
                    INSERT INTO students(
                        name,class_name,team,homeroom_teacher,
                        parent_name,parent_email,
                        student_username,student_password_hash,student_password_display,
                        status,created_at
                    ) VALUES(?,?,?,?,?,?,?,?,?,'pending',?)
                """,(
                    values[0],values[3],values[4],values[5],
                    values[6],values[7],
                    values[1],hash_pw(values[2]),values[2],now()
                ))
                con.commit()
            except Exception as e:
                con.rollback()
                messagebox.showerror("Đăng ký thất bại",f"Không thể lưu hồ sơ học sinh.\n\n{e}")
                return
            finally:
                con.close()

            messagebox.showinfo(
                "Đã gửi",
                "Hồ sơ học sinh đã gửi giáo viên phê duyệt.\n"
                "Tên đăng nhập và mật khẩu bạn vừa tạo sẽ được giữ nguyên sau khi được duyệt."
            )
            self.show_login()

        self.button(card,"📨 GỬI GIÁO VIÊN PHÊ DUYỆT",submit,30,COLORS["green"]).grid(
            row=len(labels),column=0,columnspan=2,pady=13
        )
        self.button(self,"Quay lại",self.show_login,15,"#7f8c8d").pack()

    def pending_parents(self):
        self.clear()
        teacher_code=get_teacher_code()
        self.header(
            "👨‍👩‍👧 XÁC NHẬN PHỤ HUYNH",
            f"Giáo viên kiểm tra → xác nhận phụ huynh → tạo mã + QR | Mã giáo viên: {teacher_code}"
        )
        info=tk.Frame(self,bg="#eef5ff",bd=1,relief="solid")
        info.pack(fill="x",padx=28,pady=(0,10))
        tk.Label(info,text=f"🔐 MÃ XÁC NHẬN CỦA GIÁO VIÊN:  {teacher_code}",
                 bg="#eef5ff",fg="#173f7a",font=("Arial",15,"bold")).pack(side="left",padx=16,pady=10)
        self.button(info,"🔄 Tạo mã giáo viên mới",self.teacher_code_dialog,24,COLORS["purple"]).pack(side="right",padx=10,pady=6)

        con=connect()
        rows=con.execute("SELECT * FROM parent_requests WHERE status='pending' ORDER BY id DESC").fetchall()
        con.close()
        tr=self.tree(
            ("id","parent","email","child","class","team","teacher"),
            ("ID","Tên phụ huynh","Email","Tên con","Lớp","Tổ","Giáo viên chủ nhiệm"),
            [55,180,250,180,90,70,220]
        )
        for r in rows:
            tr.insert("", "end", values=(r["id"],r["parent_name"],r["parent_email"],r["student_name"],r["class_name"],r["team"],r["homeroom_teacher"]))

        def delete_request_and_accounts():
            sel=tr.selection()
            if not sel:
                messagebox.showwarning("Chọn","Hãy chọn yêu cầu phụ huynh cần xóa.")
                return
            rid=int(tr.item(sel[0])["values"][0])
            con2=connect()
            try:
                req=con2.execute("SELECT * FROM parent_requests WHERE id=?",(rid,)).fetchone()
                if not req:
                    messagebox.showerror("Lỗi","Không tìm thấy yêu cầu.")
                    return

                # Tìm học sinh theo tên + lớp, sau đó xóa cả tài khoản HS và PH.
                student=con2.execute(
                    "SELECT id,name FROM students WHERE name=? AND class_name=? ORDER BY id DESC LIMIT 1",
                    (req["student_name"],req["class_name"])
                ).fetchone()

                confirm=messagebox.askyesno(
                    "⚠ XÓA CẢ HỌC SINH + PHỤ HUYNH",
                    f"Xóa yêu cầu của phụ huynh {req['parent_name']} và học sinh {req['student_name']}?\n\n"
                    "Hệ thống sẽ xóa cả tài khoản học sinh, tài khoản phụ huynh và yêu cầu xác nhận liên quan."
                )
                if not confirm:
                    return

                # Xóa dữ liệu liên quan trước để tránh tài khoản mồ côi.
                if student:
                    sid=student["id"]
                    for table in ("parents","class_officers"):
                        try:
                            con2.execute(f"DELETE FROM {table} WHERE student_id=?",(sid,))
                        except sqlite3.OperationalError:
                            pass
                    for table in ("scores","task_completions"):
                        try:
                            con2.execute(f"DELETE FROM {table} WHERE student_id=?",(sid,))
                        except sqlite3.OperationalError:
                            pass
                    try:
                        con2.execute("DELETE FROM students WHERE id=?",(sid,))
                    except sqlite3.OperationalError:
                        pass

                con2.execute("DELETE FROM parent_requests WHERE id=?",(rid,))
                con2.commit()
                messagebox.showinfo("Đã xóa","Đã xóa cả hồ sơ và tài khoản học sinh/phụ huynh liên quan.")
            except Exception as exc:
                con2.rollback()
                messagebox.showerror("Lỗi xóa",f"Không thể xóa hồ sơ:\n{exc}")
            finally:
                con2.close()

            self.pending_parents()

        def approve():
            sel=tr.selection()
            if not sel:
                messagebox.showwarning("Chọn","Hãy chọn yêu cầu phụ huynh."); return
            rid=tr.item(sel[0])["values"][0]
            con2=connect()
            req=con2.execute("SELECT * FROM parent_requests WHERE id=?",(rid,)).fetchone()
            student=con2.execute(
                "SELECT * FROM students WHERE name=? AND class_name=? ORDER BY id DESC LIMIT 1",
                (req["student_name"],req["class_name"])
            ).fetchone()
            if not student:
                con2.close()
                messagebox.showerror("Không tìm thấy","Không tìm thấy học sinh khớp tên và lớp. Giáo viên hãy kiểm tra lại.")
                return
            access_code=student["access_code"] or make_code()
            if student["status"] != "approved":
                con2.execute("UPDATE students SET status='approved',access_code=?,approved_at=? WHERE id=?",
                             (access_code,now(),student["id"]))
            parent_username=req["parent_username"] or req["parent_email"]
            parent_password_hash=req["parent_password_hash"]
            parent_password_display=req["parent_password_display"]
            if not parent_password_hash:
                con2.close()
                messagebox.showerror(
                    "Thiếu tài khoản phụ huynh",
                    "Phụ huynh chưa đăng ký tên đăng nhập và mật khẩu.\n"
                    "Yêu cầu này không thể cấp tài khoản."
                )
                return

            con2.execute(
                """INSERT OR REPLACE INTO parents(
                    student_id,email,username,password_hash,password_display,must_change,verified
                ) VALUES(?,?,?,?,?,?,0)""",
                (student["id"],req["parent_email"],parent_username,parent_password_hash,parent_password_display,0)
            )
            parent_temp=parent_password_display
            con2.execute("UPDATE parent_requests SET status='approved',approved_at=? WHERE id=?",(now(),rid))
            con2.commit()
            student2=con2.execute("SELECT * FROM students WHERE id=?",(student["id"],)).fetchone()
            con2.close()
            self.show_parent_confirmation_package(dict(student2),req["parent_name"],req["parent_email"],parent_temp,teacher_code,access_code)
            self.pending_parents()

        # Thanh nút cố định ở đáy: luôn nhìn thấy dù bảng dữ liệu chiếm toàn màn hình.
        actions=tk.Frame(self,bg=COLORS["bg"])
        actions.pack(side="bottom",fill="x",padx=20,pady=(5,10))
        self.button(actions,"✅ XÁC NHẬN PHỤ HUYNH + CẤP MK + TẠO QR",
                     approve,48,COLORS["green"]).pack(side="left",padx=6,ipady=2)
        self.button(actions,"🗑 XÓA CẢ HS + PH",
                     delete_request_and_accounts,22,COLORS["red"]).pack(side="left",padx=6,ipady=2)
        self.button(actions,"Quay lại",self.teacher_dashboard,15,"#7f8c8d").pack(side="right",padx=6,ipady=2)
